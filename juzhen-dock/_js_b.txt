function parentOf(p){
  const parts = String(p).split(/[\\/]+/).filter(Boolean);
  if (parts.length < 2) return String(p);
  const win = String(p).indexOf('\\') >= 0;
  return (win ? '' : '/') + parts.slice(0, -1).join(win ? '\\' : '/');
}

}

}

function parseUrl(raw){
  let u; try { u = new URL(String(raw).trim()); } catch(e){ return null; }
  if (!/^https?:$/.test(u.protocol)) return null;
  const host = u.hostname.toLowerCase();
  const pf = PLATFORMS.find(p => p.re.test(host)) || FALLBACK_PF;
  const seg = u.pathname.split('/').filter(Boolean);
  let kind = pf.kind, title = '';
  if (pf.id === 'github' && seg.length >= 2){
    title = seg[0] + ' / ' + seg[1];
    kind = seg[2] === 'issues' ? '议题' : (seg[2] === 'blob' || seg[2] === 'tree') ? '代码' : '仓库';
  } else if (pf.id === 'arxiv' && seg.length){ title = 'arXiv ' + seg[seg.length - 1]; }
  else if (pf.id === 'bili' && seg[0] === 'video'){ title = 'B站视频 ' + (seg[1] || ''); }
  else if (seg.length){
    let last = seg[seg.length - 1];
    /* 路径里出现非法的百分号序列时 decodeURIComponent 会抛 URIError，
       宁可退回原文，也不要让整条收藏流程中断。 */
    try { last = decodeURIComponent(last); } catch(e){}
    title = last.replace(/[-_]+/g, ' ').slice(0, 64);
  }
  if (!title) title = host;
  const hay = (u.pathname + ' ' + u.search).toLowerCase();
  const tags = [];
  TAG_RULES.forEach(r => { if (r.re.test(hay)) r.tags.forEach(t => { if (tags.indexOf(t) < 0) tags.push(t); }); });
  if (!tags.length) tags.push(pf.name);
  return { url:u.href, pid:pf.id, name:pf.name, ab:pf.ab, slot:pf.slot, kind, title, desc:'', tags, time:'刚刚' };
}

function codeLang(s){
  if (/^\s*(#include|int\s+main|void\s+\w+\s*\()/m.test(s)) return 'c / c++';
  if (/^\s*(def\s|import\s+\w|from\s+\w+\s+import)/m.test(s)) return 'python';
  if (/^\s*(function|const|let|var|=>|export\s|import\s)/m.test(s)) return 'javascript';
  if (/^\s*[\[{]/.test(s.trim())) return 'json';
  if (/^\s*(cd|ls|grep|git|npm|pip|docker|kubectl|sudo)\b/m.test(s)) return 'shell';
  if (/<\/?[a-zA-Z]+[^>]*>/.test(s)) return 'html';
  return '表达式';
}

function crumb(path){
  const win = String(path).indexOf('\\') >= 0;
  const parts = String(path).split(/[\\/]+/).filter(Boolean);
  return parts.map((p, i) => {
    const label = i === 0 ? (win ? p : '/' + p) : p;
    return '<span>' + esc(label) + '</span>' + (i < parts.length - 1 ? '<i>›</i>' : '');
  }).join('');
}

/* 分组 = 一块彩色大卡，分组内的条目都装在这张卡里。
   颜色由分组名派生，所以同一个分组永远是同一个颜色。
   opt.grid 用来换卡内的网格：
     · 不传  → 两列小条目网格（适合"图标 + 名称 + 副标题"的短条目）
     · 'wide' → 自适应宽格，窄面板下自动落到单列（适合收藏箱这种
                含标题/描述/标签三四层信息的内容型卡片）            */
function gcard(name, n, body, opt){
  const o = opt || {};
  const c = o.color || colorFor(name);
  const cls = 'gcard-b' + (o.one ? ' one' : '') + (o.grid ? ' ' + o.grid : '');
  /* 拖拽排序容器（R24）：opt.dragsort 传一个键名，卡身挂 data-dragsort，
     组内条目挂 data-dsid —— 全站统一的拖拽重排协议在 _js_c 的
     dragstart/dragover/drop 三条委托里，落库回调登记在 DRAG_CB[键]。 */
  const dsort = o.dragsort ? '" data-dragsort="' + esc(o.dragsort) : '';
  /* 分组头右侧那个「+」：往**当前这一个分组里**加一条。
     起因是"分组"只做了归类，没有做归入 —— 文件夹按父目录自动分成了
     Thermal / Calibration / Docs 几组，但整页只有一个"新增文件夹"，
     想把一条加进指定分组全靠运气。现在每一组自带入口。
     动作名仍然登记在 GROUP_ADD 表里（不是现场拼的）—— 现场拼出来的
     data-act 静态 HTML 里扫不到，构建期守卫就会漏掉这一整条链路，
     页头「+」已经在这上面栽过一次。
     o.addCtx 里的键会原样变成 data-* 传到按钮上（如 data-group），
     点击时再用 dataset 读回来，这样"加进哪一组"的上下文不会丢。 */
  let add = '';
  if (o.addKind){
    const act = GROUP_ADD[o.addKind];
    if (act){
      const cx = o.addCtx || {};
      const attrs = Object.keys(cx).map(k => ' data-' + k + '="' + esc(String(cx[k])) + '"').join('');
      add = '<button class="gcard-add" data-act="' + act + '"' + attrs
          + ' title="加进「' + esc(name) + '」">' + IC('plus', 13) + '</button>';
    }
  }
  return '<section class="gcard" style="' + vars(c) + '">'
       + '<div class="gcard-h"><span class="gcard-ico">' + IC(o.icon || 'folder', 18) + '</span>'
       + '<span class="gcard-nm">' + esc(name) + '</span>'
       + (o.hideN ? '' : '<span class="gcard-n">' + n + ' 项</span>')
       + add
       + '</div><div class="' + cls + dsort + '">' + body + '</div></section>';
}
function groupBy(items, keyFn){
  const m = new Map();
  items.forEach(it => { const k = keyFn(it); if (!m.has(k)) m.set(k, []); m.get(k).push(it); });
  return Array.from(m.entries()).map(e => ({ name:e[0], items:e[1] }));
}

      '<div class="row" data-kind="folder" data-path="' + esc(f.path) + '"'

      + '<div class="tx"><div class="nm">' + esc(f.name) + '</div><div class="hs">' + esc(parentOf(f.path)) + '</div></div>'
      + '<button class="ctx-more" data-act="ctx-open" title="编辑 / 删除">' + IC('more', 13) + '</button>'
      + '</div>').join('');

    const dir = parentOf(g.items[0].path);

     整页一个"手动加"都没有 —— 想补一条早先收藏的文章，没有地方可下手。
     恢复默认那条路（设置页）也确实能填回来，但那是"清空再灌演示数据"，
     不能当新增用。 */
  const addRow = '<div class="acts-row"><button data-act="clip-new">'
    + IC('plus', 14) + '新增收藏</button></div>';
  if (!list.length) return html + emptyBox(state.query ? '没有匹配的收藏'
    : (state.clipFilter === '全部' ? '收藏箱还是空的\n把链接粘到上面那一栏试试' : '这个平台下暂时没有收藏')) + addRow;

  /* 收藏箱条目的信息分层：
       首行 = 平台方块 + 平台名 + 内容类型
       次行 = 标题（2 行截断）
       再次 = 描述（2 行截断）
       底行 = 标签（左） + 时间（右，margin-left:auto）
     两个刻意的处理：
       · 时间不再跟平台名挤在首行 —— 旧版时间在首行右对齐，正好被右上角
         的删除按钮盖住，hover 时两者重叠。
       · 标签超过 3 个折叠成 +N，避免标签换行把卡片撑得高矮不一。
     外层走 gcard(..., {grid:'wide'})：面板 580px 下单列，正文约 400px
     （旧版两列只有约 154px，这是收藏箱"布局不友好"的主因）。 */
  const card = (it, i) => {
    const tags = it.tags || [];
    const shown = tags.slice(0, 3).map(t => '<span class="tg">' + esc(t) + '</span>').join('')
      + (tags.length > 3 ? '<span class="tg more">+' + (tags.length - 3) + '</span>' : '');
    return '<div class="cclip" data-kind="clip" data-url="' + esc(it.url) + '"'
      + ' style="' + vars(ACC[it.slot]) + ';--i:' + Math.min(i, 9) + '"' + stg() + '>'
      + '<div class="hd"><span class="mono">' + esc(it.ab) + '</span>'
      + '<span class="src">' + esc(it.name) + '</span>'
      + '<span class="kind">' + esc(it.kind) + '</span></div>'
      + '<div class="ti">' + esc(it.title) + '</div>'
      + (it.desc ? '<div class="ds">' + esc(it.desc) + '</div>' : '')
      + '<div class="ft"><div class="tgs">' + shown + '</div>'
      + '<span class="tm">' + esc(it.time) + '</span>'
      /* 右上角那个 x 只管删除，改标题 / 备注 / 标签没有入口。
         ⋯ 挂在底行右侧，不跟 x 抢同一个角（旧版时间被删除按钮盖住过）。 */
      + '<button class="ctx-more" data-act="ctx-open" title="编辑 / 删除">' + IC('more', 12) + '</button></div>'
      + '<button class="x" data-act="clip-del" data-url="' + esc(it.url) + '" title="移除">' + IC('x', 11) + '</button>'
      + '</div>';
  };

  /* 平台筛选态下只有一张卡（就是那个平台），也给它挂上「+」——
     否则筛到某个平台想再补一条，还得先切回"全部"。 */
  if (state.clipFilter !== '全部')
    return html + gcard(state.clipFilter, list.length, list.map(card).join(''),
      { icon:'bookmark', grid:'wide', addKind:'clip', addCtx:{ group:state.clipFilter } }) + addRow;
  let k = 0;
  html += groupBy(list, c => c.name).map(g =>
    gcard(g.name, g.items.length, g.items.map(c => card(c, k++)).join(''),
      { icon:'bookmark', grid:'wide', addKind:'clip', addCtx:{ group:g.name } })).join('');
  return html + addRow;
}

/* ---------- 页 4 · 暂存区（拟物纸质文件夹） ---------- */
function buildStaging(){
  const p = PAGE_MAP.staging;
  const list = pick(DATA.staging, ['name','path']);
  let html = pageHead(p, list.length === DATA.staging.length
    ? '引用式记录 · 共 ' + DATA.staging.length + ' 个文件'
    : '筛选命中 ' + list.length + ' / ' + DATA.staging.length + ' 个文件');
  if (!list.length) return html + '<div class="bd-empty">暂存区是空的\n把文件拖到这个面板里试试</div>';
  const cards = list.map((f, k) => {
    const e = extOf(f.name);
    const glyph = { sheet:'sheet', doc:'file', pdf:'file', slide:'slide', zip:'zip', img:'image', model:'model' }[fileKind(f.name)];
    return '<div class="fcard" data-kind="file" data-path="' + esc(f.path) + '" draggable="true"'
      + ' style="' + vars() + ';--i:' + Math.min(k, 9) + '"' + stg() + '>'
      + '<button class="fcard-nt" data-act="file-del" data-path="' + esc(f.path) + '" title="移出暂存">' + IC('x', 11) + '</button>'
      + '<div class="fcard-ic"><span class="ic">' + IC(glyph, 24) + '</span></div>'
      + '<div class="fcard-bot"><div class="fcard-nm">' + esc(f.name) + '</div>'
      + '<div class="fcard-meta"><span class="ext">' + esc((e || 'file').toUpperCase()) + '</span>'
      + '<span class="dot"></span><span>' + fmtSize(f.size) + '</span>'
      + '<span class="tag">' + esc(f.mode) + '</span>'
      /* 暂存项的名字和路径都能改（文件挪了位置，用户比我们清楚），
         但原先只有右上角一个"移出暂存"，改的入口藏在右键里。 */
      + '<button class="ctx-more" data-act="ctx-open" title="编辑 / 移出">' + IC('more', 12) + '</button></div></div></div>';
  }).join('');
  return html + '<div class="dz-wrap"><div class="dropzone" id="dropzone"><div class="cgrid">' + cards + '</div></div></div>'
    + '<div class="dz-hint">拖入即暂存 · 点击卡片即预览 · 原文件不会被移动或删除</div>';
}
