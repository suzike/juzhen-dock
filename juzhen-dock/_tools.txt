/* ================================================================
   湿空气与热舒适计算 —— ISO 7730 / ASHRAE 55 工程实现
   ----------------------------------------------------------------
   全部是可直接用于工程计算的公式，不是占位。
   ================================================================ */
const PSY = {
  /* 饱和水汽压 kPa —— Magnus 公式，ta 单位 ℃ */
  es: t => 0.6108 * Math.exp(17.27 * t / (t + 237.3)),
  /* 水汽分压 kPa */
  pv: (ta, rh) => PSY.es(ta) * rh / 100,
  /* 含湿量 kg/kg 干空气 */
  d:  (ta, rh, p) => { const e = PSY.pv(ta, rh); return 0.622 * e / ((p || 101.325) - e); },
  /* 露点温度 ℃ */
  td: (ta, rh) => { const e = Math.max(PSY.pv(ta, rh), 1e-6); const v = Math.log(e / 0.6108);
                    return 237.3 * v / (17.27 - v); },
  /* 比焓 kJ/kg 干空气 */
  h:  (ta, rh, p) => { const dd = PSY.d(ta, rh, p); return 1.006 * ta + dd * (2501 + 1.86 * ta); },
  /* 相对湿度 %（由露点反推，用于结露风险判断） */
  rhByTd: (ta, tdv) => Math.min(100, Math.max(0, PSY.es(tdv) / PSY.es(ta) * 100))
};

/* PMV / PPD —— Fanger 模型，ISO 7730 附录 D 的实现
   服装表面温度 tcl 用不动点迭代求解，带 0.5 松弛因子避免振荡。 */
function pmvSolve(ta, tr, vel, rh, met, clo){
  const M   = met * 58.15;      /* met → W/m² */
  const W   = 0;                /* 外部做功，坐姿办公取 0 */
  const Icl = clo * 0.155;      /* clo → m²·K/W */
  const pa  = PSY.pv(ta, rh);   /* kPa */
  const fcl = Icl <= 0.078 ? 1 + 1.29 * Icl : 1.05 + 0.645 * Icl;

  const hcOf = t => Math.max(12.1 * Math.sqrt(Math.max(vel, 0.05)),
                             2.38 * Math.pow(Math.abs(t - ta), 0.25));
  /* tcl 用**二分法**而不是不动点迭代。
     不动点迭代在服装热阻大（≥1.5 clo）时残差斜率 > 1，会在两个值之间来回跳，
     实测 clo=2.0 / ta=-5℃ 时直接发散到 ±Infinity，最后 Inf-Inf 得到 NaN。
     残差 g(t) 关于 t 单调递减，二分法 60 次可精确到 1e-12，且永不发散。 */
  const g = t => {
    const h  = hcOf(t);
    const rd = 3.96e-8 * fcl * (Math.pow(t + 273, 4) - Math.pow(tr + 273, 4));
    return 35.7 - 0.028 * (M - W) - Icl * (rd + fcl * h * (t - ta)) - t;
  };
  let lo = Math.max(ta - 40, -70), hi = 50;
  for (let i = 0; i < 60; i++){ const mid = (lo + hi) / 2; if (g(mid) > 0) lo = mid; else hi = mid; }
  const tcl = (lo + hi) / 2;

  hc = hcOf(tcl);
  const rad    = 3.96e-8 * fcl * (Math.pow(tcl + 273, 4) - Math.pow(tr + 273, 4));
  const conv   = fcl * hc * (tcl - ta);
  /* 注意 pa 的单位：ISO 7730 的 PMV 方程里 pa 取 **Pa**，不是 kPa。
     （皮肤扩散项 3.05e-3·(5733−6.99(M−W)−pa) 与呼吸项 1.7e-5·M·(5867−pa)
      的常数 5733 / 5867 都是按 Pa 标定的。）用 kPa 会多算约 6 W/m² 散热，
     PMV 整体偏冷近 0.4 级。 */
  const paPa = pa * 1000;
  const L = (M - W)
    - 3.05e-3 * (5733 - 6.99 * (M - W) - paPa)
    - 0.42   * ((M - W) - 58.15)
    - 1.7e-5 * M * (5867 - paPa)
    - 0.0014 * M * (34 - ta)
    - rad - conv;
  const pmv = (0.303 * Math.exp(-0.036 * M) + 0.028) * L;
  const ppd = 100 - 95 * Math.exp(-0.03353 * Math.pow(pmv, 4) - 0.2179 * pmv * pmv);
  return { pmv:pmv, ppd:ppd, tcl:tcl, hc:hc, pa:pa, fcl:fcl };
}

function buildSnip(){
  const p = PAGE_MAP.snip;
  const list = pick(TOOL.snips, ['name','lang','g','text']);

  const card = (s, i) =>

    + ' style="--i:' + Math.min(i, 9) + '"' + stg() + '>'
    + '<div class="snip-h"><span class="snip-lang">' + esc(s.lang) + '</span>'
    + '<span class="snip-nm">' + esc(s.name) + '</span>'
    + (s.vars.length ? '<span class="snip-v">' + s.vars.length + ' 变量</span>' : '')
    + '<span class="snip-cp">' + IC('copy', 13) + '</span>'
    + '<button class="ctx-more" data-act="ctx-open" title="编辑 / 删除">' + IC('more', 13) + '</button>'
    + '</div><pre class="snip-code">' + esc(s.text) + '</pre></div>';

  /* 分组按数据里实际出现的跑，不再用一张写死的表。
     片段以前是只读的，写死三种分组没问题；一旦能编辑，用户就会自己起
     分组名（"热管理公式""台架脚本"），而写死的遍历会让新分组下的片段
     整条从页面上消失 —— 存着、搜得到，就是看不见。 */
  const groups = [];
  list.forEach(s => { if (groups.indexOf(s.g) < 0) groups.push(s.g); });
  let k = 0;
  groups.forEach(g => {
    const items = list.filter(s => s.g === g);
    if (!items.length) return;
    html += gcard(g, items.length, items.map(s => card(s, k++)).join(''),
      { icon: g === '终端命令' ? 'terminal' : g === '文本模板' ? 'note' : 'code',
        one:true,
        /* 分组名直接预填进表单 —— 片段的分组是数据里的一个字段，
           不像文件夹那样是派生的，所以这是真正意义上的"加进这一组"。 */
        addKind:'snip', addCtx:{ group:g } });
  });
  return html + addRow;
}

/* 换算 / 湿空气 / PMV 的结果渲染 —— 输入变了只改这三块，不整页重建 */
function calcUpdate(){
  const g   = convGroup(state.convGroup);
  const raw = parseFloat((document.querySelector('[data-conv="val"]') || {}).value);
  const v   = isFinite(raw) ? raw : 0;
  const out = $('convOut');
  if (out && g.u[state.convFrom]){
    const base = g.u[state.convFrom].to(v);
    out.innerHTML = Object.keys(g.u).filter(u => u !== state.convFrom).map(u => {
      const r = g.u[u].from(base);
      return '<div class="cvo"><span class="cvo-v">' + fmtNum(r) + '</span><span class="cvo-u">' + esc(u) + '</span></div>';
    }).join('');
  }

  const taEl = $('psy-ta'), rhEl = $('psy-rh'), pOut = $('psyOut');
  if (taEl && rhEl && pOut){
    const ta = parseFloat(taEl.value), rh = Math.min(100, Math.max(1, parseFloat(rhEl.value)));
    if (isFinite(ta) && isFinite(rh)){
      state.psy.ta = ta; state.psy.rh = rh;
      const td  = PSY.td(ta, rh), dd = PSY.d(ta, rh) * 1000, hh = PSY.h(ta, rh), pv = PSY.pv(ta, rh);
      const fog = ta <= td + 0.5;
      pOut.innerHTML = resCell('露点温度', td.toFixed(1) + ' ℃', fog ? '结露风险' : '安全', fog)
        + resCell('含湿量', dd.toFixed(2) + ' g/kg', '干空气', false)
        + resCell('比焓', hh.toFixed(1) + ' kJ/kg', '干空气', false)
        + resCell('水汽分压', pv.toFixed(3) + ' kPa', '饱和 ' + PSY.es(ta).toFixed(3), false);
    }
  }

  const ids = ['pmv-ta','pmv-tr','pmv-vel','pmv-rh','pmv-met','pmv-clo'];
  const els = ids.map(i => $(i)), mo = $('pmvOut');
  if (els.every(Boolean) && mo){
    const ta = parseFloat(els[0].value), tr = parseFloat(els[1].value), vel = parseFloat(els[2].value);
    const rh = parseFloat(els[3].value), met = parseFloat(els[4].value), clo = parseFloat(els[5].value);
    if ([ta,tr,vel,rh,met,clo].every(isFinite)){
      state.pmvIn = { ta:ta, tr:tr, vel:vel, rh:rh, met:met, clo:clo };
      const r = pmvSolve(ta, tr, vel, rh, met, clo);
      const ok = Math.abs(r.pmv) <= 0.5;
      mo.innerHTML = '<div class="pmv-big' + (ok ? ' ok' : '') + '">'
        + '<div class="pmv-n">' + r.pmv.toFixed(2) + '</div>'
        + '<div class="pmv-l">PMV · ' + esc(pmvWord(r.pmv)) + '</div></div>'
        + resCell('PPD', r.ppd.toFixed(1) + ' %', r.ppd <= 10 ? '达标' : '超标', r.ppd > 10)
        + resCell('服装表面温度 tcl', r.tcl.toFixed(1) + ' ℃', '', false)
        + resCell('对流换热系数 hc', r.hc.toFixed(2) + ' W/(m²·K)', '', false)
        + resCell('服装面积系数 fcl', r.fcl.toFixed(3), '', false)
        + resCell('水汽分压 pa', r.pa.toFixed(3) + ' kPa', '', false);
    }
  }
}
function resCell(label, val, note, warn){
  return '<div class="rc' + (warn ? ' warn' : '') + '"><span class="rc-l">' + esc(label) + '</span>'
    + '<span class="rc-v">' + esc(val) + '</span>'
    + (note ? '<span class="rc-n">' + esc(note) + '</span>' : '') + '</div>';
}
const fmtNum = v => {
  if (!isFinite(v)) return '—';
  const a = Math.abs(v);
  if (a !== 0 && (a < 0.01 || a >= 1e6)) return v.toExponential(3);
  return Math.abs(v - Math.round(v)) < 1e-9 ? String(Math.round(v)) : v.toFixed(Math.min(4, Math.max(2, 6 - String(Math.floor(a)).length)));
};

/* 速问页。四个层次，从「能不能用」到「用了什么」：
     1) 状态条：服务商 / 模型 / Key 三样各是什么、缺哪一样，直接给「去设置」。
     2) 会话条：这一段对话叫什么、第几轮、快还是深、要不要读知识库 ——
        三个开关都记在**会话**上，不是全局设置（这一问要不要看我的资料
        是每一段对话各自的事）。
     3) 会话列表 + 本轮对话：会话是真实集合，能改名、能删、能搜；
        同一段里的问答按时间正序排，跟人读对话的方向一致。
     4) 常用问法与知识库：问法可增删改；知识库列源、报块数、能增量重扫。

   浏览器原型里知识库那张卡只写清「做不到、为什么」，**不放按钮** ——
   它需要在用户机器上读文件、切块、调嵌入模型，浏览器里没有这一层。 */
function buildAsk(){
  const p = PAGE_MAP.ask;
  const ready = aiReady();
  const sess = askSessCur();
  const turns = sess ? askTurns(sess.id) : [];
  let html = pageHead(p, TOOL.askSess.length
    ? '共 ' + TOOL.askSess.length + ' 段会话 · ' + TOOL.asks.length + ' 条问答'
    : '唤出即问 · 回答可存入速记或片段');

  html += aiStatusStrip();
  html += askSessBar(sess, turns.length);

  html += '<div class="ask-bar"><textarea id="askInput" rows="2" placeholder="'
    + (ready.ok ? '问点什么…… 当前模型 ' + esc(state.settings.ai.model)
               : '问点什么…… 先在设置页配好模型（上面的状态条里有入口）')
    + '　Ctrl + Enter 发送"></textarea>'
    + '<button class="ask-go' + (ready.ok ? '' : ' off') + '" data-act="ask-send" title="发送">'
    + IC('sparkle', 15) + '</button></div>';

  /* ---- 会话列表。当前那段用 on 标出来；每一行都带可见 ⋯（切 / 改名 / 删）。
     「新会话」既是上面那条里的按钮，也在这一组的 + 上（同一条路）。 ---- */
  const sessRows = TOOL.askSess.map((s, i) => {
    const n = askTurns(s.id).length;
    return '<div class="sess-row' + (sess && s.id === sess.id ? ' on' : '') + '"'
      + ' data-kind="askSess" data-id="' + esc(s.id) + '" style="--i:' + Math.min(i, 9) + '"' + stg() + '>'
      + '<button class="sess-b" data-act="ask-sess-use" data-id="' + esc(s.id) + '">'
      + '<span class="sess-t">' + esc(s.title || '未命名会话') + '</span>'
      + '<span class="sess-s">' + esc(s.at || '') + ' · ' + (n ? n + ' 轮' : '还没问过')
      + ' · ' + (s.mode === 'deep' ? '深' : '快')
      + (s.kb ? ' · 知识库' : '') + (s.only ? ' · 仅资料' : '') + '</span></button>'
      + '<button class="ctx-more" data-act="ctx-open" title="切到 / 重命名 / 删除">' + IC('more', 12) + '</button>'
      + '</div>';
  }).join('');
  html += gcard('会话', TOOL.askSess.length,
    sessRows || emptyBox('还没有会话\n问一句就会自动开一段；也可以点这一组的 + 先起一段'),
    { icon:'bot', hideN:true, addKind:'askSess' });

  /* ---- 本轮对话。这一块就是「多轮」看得见的地方：同一段会话里的问答按先后
     排下来，发下一句时会带着这几轮当上下文（最近 6 轮）。 ---- */
  const cards = turns.map((a, i) => askCard(a, i)).join('');
  html += gcard(sess ? ('本轮对话 · ' + (sess.title || '未命名会话')) : '本轮对话', turns.length,
    cards || emptyBox(ready.ok
      ? '这一段还没问过\n在上面输入框里问一句，它就会出现在这里'
      : '这一段还没问过\n先到设置页配好模型（状态条上有入口）'),
    { icon:'text', one:true, addKind:'ask', addCtx:{ sess:(sess ? sess.id : '') } });

  html += askTplCard();
  html += kbCard();

  const demoN = TOOL.asks.filter(a => a.state === 'demo').length;
  html += '<div class="acts-row">'
    + '<button data-act="ask-new">' + IC('plus', 14) + '手工记一条问答</button>'
    + (demoN ? '<button data-act="ask-clear-demo">' + IC('x', 14) + '清掉 ' + demoN + ' 条演示问答</button>' : '')
    + '<button data-act="ask-clear">' + IC('trash', 14) + '清空问答记录</button>'
    + '</div>';
  return html;
}
/* 会话条：这一段叫什么 + 三个开关。档位用两个按钮而不是 data-seg ——
   data-seg 会把值写进 state.settings（见 setSetting），而这三个开关都属于
   会话、不属于全局设置，混进去会让「切了会话，别的对话行为也变了」。 */
function askSessBar(sess, n){
  const mode = sess ? (sess.mode === 'deep' ? 'deep' : 'fast') : 'fast';
  return '<div class="asess">'
    + '<div class="asess-h">'
    + '<span class="asess-t">' + esc(sess ? (sess.title || '未命名会话') : '还没有会话') + '</span>'
    + '<span class="asess-n">' + (n ? '下一轮是第 ' + (n + 1) + ' 轮' : '还没开始') + '</span>'
    + (sess ? '<button class="ctx-more" data-act="ctx-open" data-kind="askSess" data-id="' + esc(sess.id) + '"'
        + ' title="切到 / 重命名 / 删除">' + IC('more', 12) + '</button>' : '')
    + '</div>'
    + '<div class="asess-r">'
    + '<button class="asess-tg' + (mode === 'fast' ? ' on' : '') + '" data-act="ask-mode"'
    + ' data-mode="fast" title="快：按设置里填的上限直接答">' + IC('play', 13) + '快</button>'
    + '<button class="asess-tg' + (mode === 'deep' ? ' on' : '') + '" data-act="ask-mode"'
    + ' data-mode="deep" title="深：开思考，输出上限自动放宽">' + IC('layers', 13) + '深</button>'
    + '<button class="asess-tg' + (sess && sess.kb ? ' on' : '') + '" data-act="ask-kb">'
    + IC('book', 13) + '知识库</button>'
    + '<button class="asess-tg' + (sess && sess.only ? ' on' : '') + '" data-act="ask-only">'
    + IC('lock', 13) + '只依据知识库</button>'
    + '<button class="asess-tg" data-act="ask-sess-new">' + IC('plus', 13) + '新会话</button>'
    + '</div>'
    + '<div class="asess-hint">' + askSessHint(sess) + '</div>'
    + '</div>';
}
/* 这一段现在是怎么设置的，用人话写出来。★ 那句是重点：开了知识库却用不了时，
   必须当场说，而不是等用户点了提问、拿到一个「没检索到」再自己猜。 */
function askSessHint(sess){
  if (!sess) return '还没有会话 —— 问一句就会自动开一段。';
  const bits = [];
  bits.push(sess.mode === 'deep'
    ? '「<b>深</b>」：会开思考、输出上限自动放宽到 3000（DeepSeek 推理型 8000），慢一些但更完整'
    : '「<b>快</b>」：按设置里填的回答长度上限直接答（默认 900）');
  if (!sess.kb) bits.push('知识库关着：它只用模型自己的知识答');
  else if (sess.only) bits.push('知识库开着且锁死：资料里没有的它会直接说不确定，不拿自己的知识补');
  else bits.push('知识库开着：每次提问先在你的资料里检索，回答里会用（片段N）标出依据');
  bits.push('多轮：最近 6 轮会作为上下文一起发出去');
  const un = sess.kb ? kbUnavailable() : '';
  if (un) bits.push('<b>★ 但知识库现在用不了：</b>' + esc(un));
  return bits.join('；');
}
/* 一条问答。档位、引用都挂在它自己身上 —— 这样才能「这一条当时是怎么问的」
   一目了然，而不是全页共用一套显示状态。 */
function askCard(a, i){
  const refs = (a.refs || []).length
    ? '<div class="qa-refs">' + a.refs.map(r =>
        '<span class="qa-ref" title="' + esc(r.source || '') + '"><i>片段' + r.n + '</i>'
        + '<span>' + esc(r.title || r.source || '未命名来源') + '</span></span>').join('') + '</div>'
    : '';
  return '<div class="qa" data-kind="ask" data-id="' + esc(a.id) + '"'
    + ' style="--i:' + Math.min(i, 9) + '"' + stg() + '>'
    + '<div class="qa-q"><span class="qa-av">' + IC('bot', 14) + '</span><span>' + esc(a.q) + '</span>'
    + (a.state === 'demo' ? '<span class="qa-demo">演示</span>' : '')
    + (a.mode === 'deep' ? '<span class="qa-chip">深</span>' : '')
    + '<button class="ctx-more" data-act="ctx-open" title="重新问 / 编辑 / 删除">' + IC('more', 12) + '</button>'
    + '</div>'
    + '<div class="qa-a' + (a.state === 'run' ? ' run' : a.state === 'err' ? ' err' : '') + '">'
    + esc(a.a) + '</div>'
    + refs
    + '<div class="qa-f"><button data-act="ask-to-note" data-id="' + esc(a.id) + '">' + IC('note', 13) + '存入速记</button>'
    + '<button data-act="ask-copy" data-id="' + esc(a.id) + '">' + IC('copy', 13) + '复制回答</button>'
    + (a.state === 'err' ? '<button data-act="ask-retry" data-id="' + esc(a.id) + '">' + IC('swap', 13) + '重试</button>' : '')
    + '<span class="qa-m">' + esc(a.at || '') + (a.model ? ' · ' + esc(a.model) : '') + '</span>'
    + '</div></div>';
}
function askTplCard(){
  const tpl = TOOL.askTpl.map((t, i) =>
    '<div class="ask-tpl" data-kind="askTpl" data-id="' + esc(t.id) + '"'
    + ' style="--i:' + Math.min(i, 9) + '"' + stg() + '>'
    + '<button class="ask-tpl-b" data-act="ask-tpl" data-id="' + esc(t.id) + '">'
    + IC(t.ic, 14) + '<span>' + esc(t.t) + '</span></button>'
    + '<button class="ctx-more" data-act="ctx-open" title="编辑 / 删除">' + IC('more', 12) + '</button>'
    + '</div>').join('');
  return gcard('常用问法', TOOL.askTpl.length, tpl || emptyBox('还没有常用问法'),
    { icon:'sparkle', hideN:true, addKind:'askTpl' });
}
/* 知识库那一栏。浏览器原型里**不摆按钮**：它需要在用户机器上读文件、
   切块、调嵌入模型 —— 摆一排点不动的按钮，比缺功能更让人困惑。 */
function kbCard(){
  if (!DESK || !JZ || !JZ.kb)
    return gcard('知识库', 0,
      '<div class="kb-note">' + IC('book', 15) + '<div><b>知识库要桌面版</b>'
      + '<span>它要在你这台机器上读文件、按语义切块、调用嵌入模型，再把向量存到本机。'
      + '浏览器原型里没有这一层，所以这里不放按钮 —— 摆了也点不动。</span></div></div>',
      { icon:'book', hideN:true });

  const un = kbUnavailable();
  const st = kbState.stats;
  const rows = (kbState.sources || []).map(s =>
    '<div class="kb-src">'
    + '<span class="kb-ic">' + IC(kbKindIcon(s.kind), 13) + '</span>'
    + '<div class="kb-m"><b>' + esc(s.label || s.target) + '</b>'
    + '<span>' + esc(kbKindName(s.kind)) + ' · ' + esc(s.target) + '</span></div>'
    + '<span class="kb-n">' + (s.chunks || 0) + ' 块</span>'
    + '<button class="mini" data-act="kb-del" data-id="' + esc(s.id) + '">移除</button>'
    + '</div>').join('');

  let body = '';
  if (kbState.error)
    body += '<div class="kb-note bad">' + IC('book', 15) + '<div><b>读取知识库状态失败了</b><span>'
      + esc(kbState.error) + '</span></div></div>';
  if (un)
    body += '<div class="kb-note">' + IC('book', 15) + '<div><b>还不能用</b><span>' + esc(un)
      + '</span></div></div>';
  body += rows || emptyBox(kbState.loading
    ? '正在读取知识库…'
    : '还没有接入任何资料\n可以加一个文件夹、几个文件、一个网页，或直接贴一段文字');
  body += '<div class="kb-acts">'
    + '<button class="mini" data-act="kb-add-folder">' + IC('folder', 13) + '加文件夹</button>'
    + '<button class="mini" data-act="kb-add-files">' + IC('file', 13) + '加文件</button>'
    + '<button class="mini" data-act="kb-add-url">' + IC('globe', 13) + '加网页</button>'
    + '<button class="mini" data-act="kb-add-text">' + IC('note', 13) + '贴一段文字</button>'
    + '<button class="mini" data-act="kb-reindex">' + IC('swap', 13) + '重新扫描</button>'
    + ((kbState.sources || []).length
        ? '<button class="mini" data-act="kb-clear">' + IC('trash', 13) + '清空知识库</button>' : '')
    + '</div>'
    + '<div class="kb-note">' + IC('book', 15) + '<div><b>'
    + (st ? (st.sources + ' 个来源 · ' + st.chunks + ' 块 · ' + (st.dim || 0) + ' 维向量')
          : '还没有向量数据')
    + '</b><span>支持 .md / .txt / 代码 / .pdf / .docx 与网页。'
    + '网页抓取是最朴素的那种（不跑 JavaScript），靠脚本渲染正文的页面会抓不到，那时会如实报错。'
    + '「重新扫描」只重嵌新增和改动过的文件，不会整库重建。</span></div></div>';
  return gcard('知识库', (kbState.sources || []).length, body, { icon:'book', hideN:true });
}
